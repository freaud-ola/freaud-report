 SELECT db_name  -- 库名
       ,tbl_name  --表名
       ,column_name  -- 字段名
       ,comment  -- 字段备注
       ,integer_idx  -- 字段顺序数字，从0开始
 from(
 select  db_name,tbl_id,tbl_name,cd_id 
 from   dws_meta.ads_hive_table_df  where dt='2026-04-06' 
 ) a left join(
  select  cd_id,comment,column_name,type_name,integer_idx
  from   dws_meta.dts_hive_columns_v2_df  
  where dt='2026-04-06' 
 ) b on a.cd_id=b.cd_id
where db_name ='dws_oversea' ;
